\sect{\'Etape 1 -- BFS ou DFS ? Un seul mot d\'ecide}

\begin{regle}{La r\`egle}
\Large Le mot \mot{niveau} (ou \mot{largeur}) est dans l'\'enonc\'e \;$\Rightarrow$\; \BFS.\\[3pt]
Sinon \;$\Rightarrow$\; \DFS{} \normalsize (c'est le choix \textbf{par d\'efaut}).\\[3pt]
\normalsize Exception : le sujet \textbf{impose} le parcours (« en profondeur ») $\Rightarrow$ on ob\'eit.
\end{regle}

\begin{multicols}{2}
\begin{boxbfs}{Mots qui veulent dire BFS}
\begin{itemize}
\item \mot{niveau}, par niveau, chaque niveau
\item \mot{largeur}
\item m\^eme \mot{profondeur que x} (sur le m\^eme niveau)
\item le \mot{plus proche de la racine}
\end{itemize}
\end{boxbfs}
\columnbreak
\begin{boxdfs}{Mots qui veulent dire DFS}
\begin{itemize}
\item \mot{fils}, \mot{p\`ere}, \mot{sous-arbre}, \mot{chemin}, \mot{feuille}
\item \mot{hauteur}, taille, somme, moyenne
\item profondeur \textbf{d'un n\oe ud} ou \textbf{des feuilles}
\item « parcours profondeur » impos\'e
\end{itemize}
\end{boxdfs}
\end{multicols}

\vspace{-4pt}
\begin{center}
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{p{9.8cm} l c}
\toprule
\textbf{\'Enonc\'e} & \textbf{Mot rep\`ere} & \textbf{Parcours}\\
\midrule
Chaque \textbf{niveau} a plus de n\oe uds que le pr\'ec\'edent & niveau & \BFS\\
Nombre max de n\oe uds sur un m\^eme \textbf{niveau} & niveau & \BFS\\
N\oe ud de cl\'e x le \textbf{plus proche de la racine} & plus proche & \BFS\\
\midrule
Cl\'e de chaque n\oe ud = somme des cl\'es de ses \textbf{fils} & fils & \DFS\\
Existe-t-il un \textbf{chemin} racine $\to$ feuille de somme s ? & chemin & \DFS\\
Profondeur moyenne des \textbf{feuilles} \hfill{\color{theme}\small(pi\`ege)} & feuilles & \DFS\\
\bottomrule
\end{tabular}
\end{center}

\begin{multicols}{2}
\textbf{Pourquoi ?} Ordre de visite sur le m\^eme arbre :
\begin{center}
\begin{forest}
for tree={circle,draw=dfs,thick,minimum size=15pt,inner sep=1pt,font=\small\bfseries\color{dfs},s sep=6pt,l sep=8pt}
[1 [2 [3] [4]] [5 [6]] [7]]
\end{forest}
\quad
\begin{forest}
for tree={circle,draw=bfs,thick,minimum size=15pt,inner sep=1pt,font=\small\bfseries\color{bfs},s sep=6pt,l sep=8pt}
[1 [2 [5] [6]] [3 [7]] [4]]
\end{forest}\\[2pt]
{\small\textcolor{dfs}{DFS} : descend d'abord \hfill \textcolor{bfs}{BFS} : niveau par niveau}
\end{center}
\columnbreak
\begin{itemize}
\item Le \BFS{} donne tous les n\oe uds d'un niveau \textbf{\`a la suite} $\to$ on compare par niveau et on \textbf{s'arr\^ete t\^ot}.
\item Le \DFS{} voit chaque n\oe ud \textbf{avec ses fils} $\to$ id\'eal pour p\`ere/fils, sous-arbres, chemins.
\end{itemize}
\end{multicols}

\begin{boxok}{\'Etape 2 -- Les 4 questions avant d'\'ecrire une ligne}
\begin{enumerate}
\item \textbf{BFS ou DFS ?} $\to$ le mot rep\`ere ci-dessus.
\item \textbf{Que renvoie une feuille ?} $\to$ fixe le cas de base et la valeur de d\'epart.
\item \textbf{Comment je combine les fils ?} $\to$ somme, \code{max}, « tous vrais » (CHAQUE), « au moins un » (IL EXISTE).
\item \textbf{Puis-je m'arr\^eter avant la fin ?} $\to$ si oui : condition dans le \code{while} ou \code{return} dans la boucle.
\end{enumerate}
\end{boxok}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l l l}
\toprule
 & \TREE & \TAB\\
\midrule
Est-ce une feuille ? & \tc{T.nbchildren == 0} & \bc{B.child is None}\\
Premier fils & \tc{T.children[0]} & \bc{C = B.child}\\
Fils suivant & \tc{i += 1} \; puis \tc{T.children[i]} & \bc{C = C.sibling}\\
Parcourir tous les fils & \tc{for C in T.children:} & \bc{while C: ... C = C.sibling}\\
Nombre de fils & \tc{T.nbchildren} & \bc{\`a compter (boucle)}\\
« Arriv\'e au bout » (boucle while) & \tc{i == T.nbchildren} & \bc{C is None}\\
« Arr\^et\'e avant la fin » & \tc{i < T.nbchildren} & \bc{C is not None}\\
\bottomrule
\end{tabular}
\end{center}

\begin{regle}{Comment lire les squelettes}
Les mots \textbf{EN MAJUSCULES} (\code{DEPART}, \code{COMBINE}, \code{FIN}, \code{LOCAL\_OK}, \code{TEST}) ne sont \textbf{pas} du Python : ce sont des \textbf{trous \`a remplir} selon l'exo. Chaque pattern est donn\'e \textcolor{treec}{\textbf{\`a gauche en Tree}} et \textcolor{tabc}{\textbf{\`a droite en TreeAsBin}}.
\end{regle}

\sect{DFS 1 -- Accumuler (taille, feuilles, hauteur)}

On appelle la fonction sur \textbf{chaque fils}, on combine, et on fait \code{return} \textbf{apr\`es} la boucle.

\begin{multicols}{2}
\TREE
\begin{lstlisting}[style=tree]
def f(T):
    acc = DEPART       # ce que vaut une feuille
    for C in T.children:   # chaque fils C
        r = f(C)       # résultat du sous-arbre C
        acc = COMBINE(acc, r)  # ex : acc + r
                               #   ou max(acc, r)
    return FIN(acc)    # souvent juste acc
\end{lstlisting}
\columnbreak
\TAB
\begin{lstlisting}[style=tab]
def f(B):
    acc = DEPART       # ce que vaut une feuille
    C = B.child        # premier fils
    while C:           # tant qu'il reste un fils
        r = f(C)       # résultat du sous-arbre C
        acc = COMBINE(acc, r)
        C = C.sibling  # fils suivant (obligatoire !)
    return FIN(acc)
\end{lstlisting}
\end{multicols}

\begin{center}
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{l l l l l}
\toprule
\textbf{Fonction} & \textbf{D\'epart} & \textbf{Combinaison} & \textbf{Fin} & \textbf{\code{if} feuille ?}\\
\midrule
taille & \code{s = 1} & \code{s += f(C)} & \code{s} & non\\
feuilles & \code{s = 0} & \code{s += f(C)} & \code{s} & oui : \code{return 1}\\
hauteur (feuille = 0) & \code{h = -1} & \code{h = max(h, f(C))} & \code{h + 1} & non\\
arit\'e max & \code{k = m = 0} & \code{k += 1 ; m = max(m, f(C))} & \code{max(k, m)} & non\\
max des fils & \code{m = 1\textsuperscript{re} cl\'e} & \code{m = max(m, C.key)} & \code{m} & -- \\
\bottomrule
\end{tabular}
\end{center}
{\small Le \code{if} feuille ne sert que si une feuille renvoie autre chose que la formule g\'en\'erale avec z\'ero fils. Pour un \textbf{max}, ne jamais partir de 0 (cl\'es n\'egatives) : partir du premier \'el\'ement r\'eel.}

\sect{DFS 2 -- V\'erifier : « CHAQUE » ou « IL EXISTE » ?}

\begin{regle}{Le mot de l'\'enonc\'e choisit la boucle}
\textbf{« chaque », « tous »} $\Rightarrow$ tous les fils doivent r\'eussir $\Rightarrow$ on s'arr\^ete au \textbf{premier \'echec}.\\
\textbf{« existe-t-il », « au moins un »} $\Rightarrow$ un seul fils suffit $\Rightarrow$ on s'arr\^ete au \textbf{premier succ\`es}.\\[2pt]
Rappel : la condition d'un \code{while} dit quand on \textbf{continue}, pas quand on s'arr\^ete.
\end{regle}

\textbf{\large CHAQUE / TOUS} \hfill {\small check\_sum, check\_max, heap\_min, even\_leaves, deep\_enough}
\begin{multicols}{2}
\TREE
\begin{lstlisting}[style=tree]
def check(T):
    if not LOCAL_OK(T):  # T respecte la règle ?
        return False
    i = 0
    while (i < T.nbchildren
           and check(T.children[i])):
        i += 1           # fils OK : au suivant
    return i == T.nbchildren  # au bout = tous OK
\end{lstlisting}
\columnbreak
\TAB
\begin{lstlisting}[style=tab]
def check(B):
    if not LOCAL_OK(B):  # B respecte la règle ?
        return False
    C = B.child
    while C and check(C):  # tant que le fils RÉUSSIT
        C = C.sibling
    return C is None     # au bout = tous OK
\end{lstlisting}
\end{multicols}

\textbf{\large IL EXISTE} \hfill {\small path\_sum, path\_increasing, path\_below, leaf\_at\_depth}
\begin{multicols}{2}
\TREE
\begin{lstlisting}[style=tree]
def exists(T):
    if not LOCAL_OK(T):    # chemin mort ?
        return False
    if T.nbchildren == 0:  # feuille atteinte
        return True        # (ou le test final)
    i = 0
    while (i < T.nbchildren
           and not exists(T.children[i])):
        i += 1             # fils échoue : au suivant
    return i < T.nbchildren  # arrêté avant = trouvé
\end{lstlisting}
\columnbreak
\TAB
\begin{lstlisting}[style=tab]
def exists(B):
    if not LOCAL_OK(B):    # chemin mort ?
        return False
    if B.child is None:    # feuille atteinte
        return True        # (ou le test final)
    C = B.child
    while C and not exists(C):  # tant que ÉCHEC
        C = C.sibling
    return C is not None   # arrêté sur un fils
\end{lstlisting}
\end{multicols}

\textbf{Versions avec \code{return} dans la boucle} (accept\'ees, souvent plus lisibles)
\begin{multicols}{2}
\TREE
\begin{lstlisting}[style=tree]
# CHAQUE                    # IL EXISTE
for C in T.children:        for C in T.children:
    if not check(C):            if exists(C):
        return False                return True
return True                 return False
\end{lstlisting}
\columnbreak
\TAB
\begin{lstlisting}[style=tab]
# CHAQUE                    # IL EXISTE
C = B.child                 C = B.child
while C:                    while C:
    if not check(C):            if exists(C):
        return False                return True
    C = C.sibling               C = C.sibling
return True                 return False
\end{lstlisting}
\end{multicols}

\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{l l l}
\toprule
 & \textbf{CHAQUE} & \textbf{IL EXISTE}\\
\midrule
Condition du \code{while} & \code{f(fils)} & \code{not f(fils)}\\
\code{return} final \TREE & \tc{i == T.nbchildren} & \tc{i < T.nbchildren}\\
\code{return} final \TAB & \bc{C is None} & \bc{C is not None}\\
Cas feuille \`a \'ecrire ? & seulement si la feuille a sa propre r\`egle & \textbf{toujours} (sinon rien n'est trouv\'e)\\
\bottomrule
\end{tabular}\\[3pt]
{\small Les deux sont \textbf{en miroir} : le \code{not} va \textbf{soit} dans le \code{while}, \textbf{soit} dans le \code{return}, jamais les deux, jamais aucun.}
\end{center}

\begin{boxok}{O\`u placer le test de la r\`egle ? « Quels n\oe uds sont concern\'es ? »}
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{l l l}
\textbf{Les feuilles seulement} & $\to$ dans le \code{if} feuille & ex. even\_leaves : \code{return T.key \% 2 == 0}\\
\textbf{Tous les n\oe uds} & $\to$ au \textbf{d\'ebut}, avant la boucle & ex. heap\_min, path\_below, deep\_enough\\
\textbf{Les n\oe uds internes} & $\to$ \textbf{apr\`es} la boucle & ex. check\_sum : \code{... and s == T.key}\\
\end{tabular}
\end{boxok}

\sect{DFS 3 -- Remonter un tuple (plusieurs infos, un seul parcours)}

\begin{multicols}{2}
\TREE
\begin{lstlisting}[style=tree]
def __aux(T):
    """renvoie (ok, nb, somme) du
       sous-arbre T"""
    (nb, s) = (1, T.key)    # T tout seul
    ok = True
    i = 0
    while i < T.nbchildren and ok:
        (ok, n, sc) = __aux(T.children[i])
        ok = ok and TEST(n, sc, T)
                            # ex : sc / n <= T.key
        nb += n             # + ses noeuds
        s += sc             # + ses clés
        i += 1
    return (ok, nb, s)

def f(T):
    return __aux(T)[0]      # on garde ok
\end{lstlisting}
\columnbreak
\TAB
\begin{lstlisting}[style=tab]
def __aux(B):
    """renvoie (ok, nb, somme) du
       sous-arbre B"""
    (nb, s) = (1, B.key)    # B tout seul
    ok = True
    C = B.child
    while C and ok:         # stop dès ok = False
        (ok, n, sc) = __aux(C)
        ok = ok and TEST(n, sc, B)
                            # ex : sc / n <= B.key
        nb += n             # + ses noeuds
        s += sc             # + ses clés
        C = C.sibling
    return (ok, nb, s)

def f(B):
    return __aux(B)[0]      # on garde ok
\end{lstlisting}
\end{multicols}
{\small Deux parcours s\'epar\'es (size puis somme) = \textbf{points perdus}. Docstring \textbf{obligatoire} : dire ce que vaut chaque case du tuple. Exemples TD : average\_subtrees $\to$ \code{(ok, nb, somme)}, average\_arity $\to$ \code{(nb internes, somme arit\'es)}.}

\sect{DFS 4 -- Un param\`etre qui descend}

Le tuple \textbf{remonte} des fils vers le p\`ere ; ici l'info \textbf{descend} du p\`ere vers les fils.

\begin{multicols}{2}
\TREE
\begin{lstlisting}[style=tree]
def __aux(T, d):
    """d = profondeur de T (racine : 0)"""
    if T.nbchildren == 0:   # feuille :
        return d            # sa profondeur
    s = 0
    for C in T.children:
        s += __aux(C, d + 1)  # fils : 1 cran plus bas
    return s

def lce(T):
    return __aux(T, 0)      # on part de la racine
\end{lstlisting}
\columnbreak
\TAB
\begin{lstlisting}[style=tab]
def __aux(B, d):
    """d = profondeur de B (racine : 0)"""
    if B.child is None:     # feuille :
        return d            # sa profondeur
    s = 0
    C = B.child
    while C:
        s += __aux(C, d + 1)  # fils : 1 cran plus bas
        C = C.sibling
    return s

def lce(B):
    return __aux(B, 0)
\end{lstlisting}
\end{multicols}

\begin{center}
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{l l l l}
\toprule
\textbf{Exo} & \textbf{Param\`etre} & \textbf{Vers le fils} & \textbf{Au d\'epart (racine)}\\
\midrule
cheminement, deep\_enough & profondeur \code{d} & \code{d + 1} & \code{0}\\
cl\'es du niveau k & niveaux restants \code{k} & \code{k - 1} & \code{k} ; on prend la cl\'e quand \code{k == 0}\\
chemin de somme s & reste \code{s} & \code{s - cl\'e} & \code{s}\\
heap\_min, path\_increasing & cl\'e du p\`ere \code{prev} & cl\'e du n\oe ud & faux p\`ere : \code{cl\'e - 1}\\
liste \`a remplir & liste \code{L} & la m\^eme \code{L} & \code{[]} (puis \code{L.append(...)})\\
\bottomrule
\end{tabular}
\end{center}

\begin{boxok}{Faut-il une fonction auxiliaire ? \'Ecris la phrase d'abord (c'est la docstring exig\'ee)}
« Vrai si \ldots{} dans le sous-arbre, \textbf{sachant que} \ldots{} »
\begin{itemize}
\item Pas de « sachant que » $\Rightarrow$ \textbf{pas d'auxiliaire} (hauteur, check\_sum, path\_sum : \code{s} est d\'ej\`a dans l'\'enonc\'e).
\item Il manque une info \textbf{d'au-dessus} (p\`ere, profondeur, reste\ldots) $\Rightarrow$ auxiliaire avec \textbf{un param\`etre qui descend}.
\item Il faut \textbf{plusieurs infos d'en dessous} (nombre ET somme) $\Rightarrow$ auxiliaire qui \textbf{renvoie un tuple}.
\item Jamais une fonction s\'epar\'ee pour « calculer la profondeur » : un n\oe ud ne conna\^it pas son p\`ere, on la fait \textbf{descendre}.
\end{itemize}
\end{boxok}

\sect{BFS -- Par niveaux (marqueur \code{None} ou 2 files)}

La file m\'elange deux niveaux. On rep\`ere la \textbf{fin d'un niveau} : c'est \textbf{le seul endroit} o\`u on compare et o\`u on s'arr\^ete.

\textbf{\large Marqueur \code{None}}
\begin{multicols}{2}
\TREE
\begin{lstlisting}[style=tree]
q = Queue()
q.enqueue(T)
q.enqueue(None)       # None = fin du niveau 0
cur = 0               # noeuds vus dans ce niveau
while not q.isempty():
    T = q.dequeue()
    if T is None:     # >>> FIN D'UN NIVEAU <<<
        ...           # ici : comparer, return...
        cur = 0
        if not q.isempty():  # sinon boucle infinie
            q.enqueue(None)
    else:
        cur += 1      # un noeud de plus
        for C in T.children:
            q.enqueue(C)     # niveau suivant
\end{lstlisting}
\columnbreak
\TAB
\begin{lstlisting}[style=tab]
q = Queue()
q.enqueue(B)
q.enqueue(None)       # None = fin du niveau 0
cur = 0
while not q.isempty():
    B = q.dequeue()
    if B is None:     # >>> FIN D'UN NIVEAU <<<
        ...           # ici : comparer, return...
        cur = 0
        if not q.isempty():
            q.enqueue(None)
    else:
        cur += 1
        C = B.child
        while C:             # enfiler les fils
            q.enqueue(C)
            C = C.sibling
\end{lstlisting}
\end{multicols}

\textbf{\large Deux files}
\begin{multicols}{2}
\TREE
\begin{lstlisting}[style=tree]
q = Queue()
q.enqueue(T)
while not q.isempty():     # 1 tour = 1 niveau
    q_next = Queue()       # niveau suivant

\sect{Parcours parall\`ele \code{Tree} / \code{TreeAsBin}}

\begin{multicols}{2}
\begin{lstlisting}
def symmetric(T, B):
    if T.key != B.key:     # clés différentes : perdu
        return False
    C = B.child            # B lu de gauche à droite
    i = T.nbchildren - 1   # T lu de droite à gauche
    while C and i >= 0 and \
          symmetric(T.children[i], C):
        i -= 1             # on avance des deux côtés
        C = C.sibling
    return C is None and i == -1
                           # les deux finis ensemble
\end{lstlisting}
\columnbreak
\begin{itemize}
\item Un \textbf{indice} \tc{i} avance dans \tc{T.children}, un \textbf{pointeur} \bc{C} avance sur les fr\`eres.
\item Fin : \textbf{les deux \'epuis\'es en m\^eme temps}.
\item \tc{Tree} $\to$ \bc{TreeAsBin} : parcourir les fils \textbf{de droite \`a gauche} et cha\^iner chaque copie devant la pr\'ec\'edente (\bc{C.sibling = first}).
\item \bc{TreeAsBin} $\to$ \tc{Tree} : \tc{T.children.append(copie(C))} pour chaque fils \bc{C}.
\end{itemize}
\end{multicols}
